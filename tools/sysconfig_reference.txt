const GPIO = scripting.addModule("/ti/driverlib/GPIO", {}, false);
function pins(name, items, input, interrupt) {
    const group = GPIO.addInstance();
    group.$name = name;
    group.associatedPins.create(items.length);
    items.forEach(function(item, i) {
        const pin = group.associatedPins[i];
        pin.$name = item[0];
        pin.pin.$assign = item[1];
        if (input) {
            pin.direction = "INPUT";
            pin.internalResistor = "PULL_DOWN";
            pin.hysteresisControl = "ENABLE";
        } else {
            pin.initialValue = item[2] ? "SET" : "CLEARED";
        }
        if (interrupt) {
            pin.interruptEn = true;
            pin.polarity = "RISE_FALL";
        }
    });
}

const I2C = scripting.addModule("/ti/driverlib/I2C", {}, false);
function i2c(name, hardware, scl, sda) {
    const bus = I2C.addInstance();
    bus.$name = name;
    bus.basicEnableController = true;
    bus.basicControllerBusSpeed = 400000;
    bus.peripheral.$assign = hardware;
    bus.peripheral.sclPin.$assign = scl;
    bus.peripheral.sdaPin.$assign = sda;
}
